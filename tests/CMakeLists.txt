cmake_minimum_required(VERSION 3.20)

add_executable(page_slide_tests page_slide_tests.cpp)
target_include_directories(page_slide_tests PRIVATE ../firmware/components/bike/include)
target_compile_features(page_slide_tests PRIVATE cxx_std_20)
if(NOT MSVC)
    target_compile_options(page_slide_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(BIKE_SANITIZERS)
        target_compile_options(page_slide_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(page_slide_tests PRIVATE -fsanitize=address,undefined)
    endif()
endif()
add_test(NAME page_slide COMMAND page_slide_tests)
